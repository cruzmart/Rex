type integer = Int;

// ------------------------------------------------
// VALID CASES
// ------------------------------------------------

function valid_simple() -> integer {
    while true {
        break;
    }
    1;
}

function valid_nested() -> integer {
    for i in 0..10 {

        while i < 5 {

            if i == 3 {
                break;
            } elifx i == 4 {
                break;
            } else {
                // still inside loop
                break;
            }

        }

        break;
    }

    0;
}

function valid_deep_nesting() -> integer {
    loop {
        if true {
            for j in 0..5 {
                if j == 2 {
                    break;
                }
            }
        }
        break;
    }
    1;
}
